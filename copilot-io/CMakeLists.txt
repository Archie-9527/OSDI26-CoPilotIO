cmake_minimum_required(VERSION 3.22)
if(NOT CMAKE_CUDA_ARCHITECTURES OR CMAKE_CUDA_ARCHITECTURES STREQUAL "native")
  message(FATAL_ERROR "Set explicit CMAKE_CUDA_ARCHITECTURES")
endif()
project(libnvm LANGUAGES C CXX CUDA)
find_package(CUDAToolkit REQUIRED)
find_package(Threads REQUIRED)
file(GLOB sources CONFIGURE_DEPENDS src/*.cpp src/linux/*.cpp)
add_library(libnvm SHARED ${sources})
target_include_directories(libnvm PUBLIC include PRIVATE src src/linux)
target_compile_definitions(libnvm PRIVATE _CUDA __CUDA__)
target_compile_features(libnvm PRIVATE cxx_std_17)
target_link_libraries(libnvm PRIVATE CUDA::cudart CUDA::cuda_driver Threads::Threads)
set_target_properties(libnvm PROPERTIES OUTPUT_NAME nvm LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
