cmake_minimum_required(VERSION 3.22)
project(swarmio_read LANGUAGES C CXX CUDA)
find_package(CUDAToolkit REQUIRED)
find_package(Threads REQUIRED)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
add_subdirectory("${ENGINE_SOURCE}" engine)
add_executable(block-benchmark-module "${BENCH_SOURCE}")
target_include_directories(block-benchmark-module PRIVATE "${ENGINE_SOURCE}/include")
target_compile_definitions(block-benchmark-module PRIVATE __CUDA__)
target_compile_options(block-benchmark-module PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo>)
target_link_libraries(block-benchmark-module PRIVATE libnvm CUDA::cudart CUDA::cuda_driver Threads::Threads)
if(ENGINE STREQUAL "copilotio")
 target_compile_definitions(block-benchmark-module PRIVATE SWARMIO_COPILOT)
 target_include_directories(block-benchmark-module PRIVATE "${GDRCOPY_SOURCE}/include")
 target_link_libraries(block-benchmark-module PRIVATE "${GDRCOPY_LIBRARY}")
endif()
set_target_properties(block-benchmark-module PROPERTIES OUTPUT_NAME nvm-block-bench RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
